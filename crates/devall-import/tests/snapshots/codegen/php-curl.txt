===== get-query =====
<?php

$ch = curl_init();

curl_setopt_array($ch, [
    CURLOPT_URL => 'https://api.example.com/v1/users?page=2&q=ada+lovelace',
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER => [
        'Accept: application/json',
        'Authorization: Bearer ••••••',
        'X-Trace: a"b\'c$d',
    ],
    CURLOPT_FOLLOWLOCATION => true,
]);

$response = curl_exec($ch);
if ($response === false) {
    echo 'Error: ' . curl_error($ch) . PHP_EOL;
} else {
    echo curl_getinfo($ch, CURLINFO_RESPONSE_CODE) . PHP_EOL;
    echo $response . PHP_EOL;
}
curl_close($ch);
===== post-json =====

    CURLOPT_URL => 'https://api.example.com/v1/users',

===== binary =====
<?php

// 4-byte binary body: save it as body.bin
$ch = curl_init();

curl_setopt_array($ch, [
    CURLOPT_URL => 'https://api.example.com/v1/blobs',
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_CUSTOMREQUEST => 'POST',
    CURLOPT_HTTPHEADER => [
        'Content-Type: application/octet-stream',
    ],
    CURLOPT_POSTFIELDS => file_get_contents('body.bin'),
    CURLOPT_FOLLOWLOCATION => true,
]);

$response = curl_exec($ch);
if ($response === false) {
    echo 'Error: ' . curl_error($ch) . PHP_EOL;
} else {
    echo curl_getinfo($ch, CURLINFO_RESPONSE_CODE) . PHP_EOL;
    echo $response . PHP_EOL;
}
curl_close($ch);

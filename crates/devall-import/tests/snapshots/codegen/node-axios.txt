===== multipart =====
import axios from "axios";
import fs from "node:fs";

const form = new FormData();
form.append("note", "hello 'world'");
form.append("file", await fs.openAsBlob("/project/files/a b.png", { type: "image/png" }), "a b.png");

const response = await axios.request({
  method: "post",
  url: "https://api.example.com/v1/upload",
  headers: {
    "X-Upload-Id": "42",
  },
  data: form,
  validateStatus: () => true,
});

console.log(response.status);
console.log(response.data);
===== options =====
import axios from "axios";
import https from "node:https";

// Digest auth (user "admin") isn't built into axios: answer the 401 challenge yourself.
const response = await axios.request({
  method: "delete",
  url: "https://internal.example.com/v1/cache",
  maxRedirects: 0,
  timeout: 2500,
  httpsAgent: new https.Agent({ rejectUnauthorized: false }),
  proxy: { protocol: "http", host: "proxy.local", port: 3128 },
  validateStatus: () => true,
});

console.log(response.status);
console.log(response.data);

===== get-query =====
import Foundation

var request = URLRequest(url: URL(string: "https://api.example.com/v1/users?page=2&q=ada+lovelace")!)
request.httpMethod = "GET"
request.addValue("application/json", forHTTPHeaderField: "Accept")
request.addValue("Bearer ••••••", forHTTPHeaderField: "Authorization")
request.addValue("a\"b'c$d", forHTTPHeaderField: "X-Trace")

let (data, response) = try await URLSession.shared.data(for: request)
if let http = response as? HTTPURLResponse {
    print(http.statusCode)
}
print(String(decoding: data, as: UTF8.self))

===== multipart =====
import Foundation

var request = URLRequest(url: URL(string: "https://api.example.com/v1/upload")!)
request.httpMethod = "POST"
request.addValue("42", forHTTPHeaderField: "X-Upload-Id")

let boundary = "Boundary-\(UUID().uuidString)"
request.setValue("multipart/form-data; boundary=\(boundary)", forHTTPHeaderField: "Content-Type")
var body = Data()
body.append(Data("--\(boundary)\r\n".utf8))
body.append(Data("Content-Disposition: form-data; name=\"note\"\r\n\r\n".utf8))
body.append(Data("hello 'world'".utf8))
body.append(Data("\r\n".utf8))
body.append(Data("--\(boundary)\r\n".utf8))
body.append(Data("Content-Disposition: form-data; name=\"file\"; filename=\"a b.png\"\r\nContent-Type: image/png\r\n\r\n".utf8))
body.append(try Data(contentsOf: URL(fileURLWithPath: "/project/files/a b.png")))
body.append(Data("\r\n".utf8))
body.append(Data("--\(boundary)--\r\n".utf8))
request.httpBody = body

let (data, response) = try await URLSession.shared.data(for: request)
if let http = response as? HTTPURLResponse {
    print(http.statusCode)
}
print(String(decoding: data, as: UTF8.self))

===== get-query =====
package main

import (
	"fmt"
	"io"
	"net/http"
)

func main() {
	req, err := http.NewRequest("GET", "https://api.example.com/v1/users?page=2&q=ada+lovelace", nil)
	if err != nil {
		panic(err)
	}
	req.Header.Add("Accept", "application/json")
	req.Header.Add("Authorization", "Bearer ••••••")
	req.Header.Add("X-Trace", "a\"b'c$d")

	client := http.DefaultClient
	res, err := client.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, err := io.ReadAll(res.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(res.Status)
	fmt.Println(string(data))
}

===== multipart =====
package main

import (
	"bytes"
	"fmt"
	"io"
	"mime/multipart"
	"net/http"
	"os"
)

func main() {
	var body bytes.Buffer
	writer := multipart.NewWriter(&body)
	if err := writer.WriteField("note", "hello 'world'"); err != nil {
		panic(err)
	}
	// "file" is sent as application/octet-stream; use writer.CreatePart for image/png
	{
		file, err := os.Open("/project/files/a b.png")
		if err != nil {
			panic(err)
		}
		part, err := writer.CreateFormFile("file", "a b.png")
		if err != nil {
			panic(err)
		}
		if _, err := io.Copy(part, file); err != nil {
			panic(err)
		}
		file.Close()
	}
	if err := writer.Close(); err != nil {
		panic(err)
	}

	req, err := http.NewRequest("POST", "https://api.example.com/v1/upload", &body)
	if err != nil {
		panic(err)
	}
	req.Header.Add("X-Upload-Id", "42")
	req.Header.Set("Content-Type", writer.FormDataContentType())

	client := http.DefaultClient
	res, err := client.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, err := io.ReadAll(res.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(res.Status)
	fmt.Println(string(data))
}

package main

import (
	"crypto/tls"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"time"
)

func main() {
	req, err := http.NewRequest("DELETE", "https://internal.example.com/v1/cache", nil)
	if err != nil {
		panic(err)
	}
	// Digest auth (user "admin"): net/http has no built-in digest support.

	client := &http.Client{
		Timeout: 2500 * time.Millisecond,
		CheckRedirect: func(req *http.Request, via []*http.Request) error {
			return http.ErrUseLastResponse
		},
		Transport: &http.Transport{
			TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
			Proxy: func(*http.Request) (*url.URL, error) { return url.Parse("http://proxy.local:3128") },
		},
	}
	res, err := client.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, err := io.ReadAll(res.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(res.Status)
	fmt.Println(string(data))
}

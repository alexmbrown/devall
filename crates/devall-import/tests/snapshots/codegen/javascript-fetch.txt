===== multipart =====
import fs from "node:fs";

const form = new FormData();
form.append("note", "hello 'world'");
form.append("file", await fs.openAsBlob("/project/files/a b.png", { type: "image/png" }), "a b.png");

const response = await fetch("https://api.example.com/v1/upload", {
  method: "POST",
  headers: {
    "X-Upload-Id": "42",
  },
  body: form,
});

console.log(response.status);
console.log(await response.text());

===== binary =====
import fs from "node:fs";

const response = await fetch("https://api.example.com/v1/blobs", {
  method: "POST",
  headers: {
    "Content-Type": "application/octet-stream",
  },
  body: await fs.openAsBlob("body.bin"), // 4-byte binary body: save it as body.bin
});

console.log(response.status);
console.log(await response.text());
